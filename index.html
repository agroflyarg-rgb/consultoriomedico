📖 SQL para Supabase

Si necesitás recrear las tablas, ejecutá este SQL:

CREATE TABLE IF NOT EXISTS pacientes ( id uuid DEFAULT gen_random_uuid() PRIMARY KEY, nombre text NOT NULL, apellido text NOT NULL, telefono text, obra_social text, created_at timestamptz DEFAULT now() ); CREATE TABLE IF NOT EXISTS turnos ( id uuid DEFAULT gen_random_uuid() PRIMARY KEY, paciente_id uuid REFERENCES pacientes(id) ON DELETE CASCADE, nombre text NOT NULL, apellido text NOT NULL, obra_social text, telefono text, fecha date NOT NULL, pago boolean DEFAULT false, monto numeric(10,2), created_at timestamptz DEFAULT now() ); CREATE TABLE IF NOT EXISTS historia_clinica ( id uuid DEFAULT gen_random_uuid() PRIMARY KEY, paciente_id uuid REFERENCES pacientes(id) ON DELETE CASCADE, nombre text NOT NULL, apellido text NOT NULL, fecha date NOT NULL, diagnostico text, tratamiento text, observaciones text, created_at timestamptz DEFAULT now() );
